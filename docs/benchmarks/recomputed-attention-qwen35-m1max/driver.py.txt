import hashlib,json,os,subprocess,sys
from pathlib import Path
root=Path('/tmp/onnax-recompute-budget-qwen35');root.mkdir(exist_ok=False)
python='/Users/justinc/Documents/GitHub/onnax/.venv/bin/python'
lib='/tmp/onnax-recompute-budget-runtime/libonnxruntime_mlx_ep.dylib'
base=[python,'-m','examples.huggingface.block_recompute','--bundle','/tmp/onnax-qwen35-complete-context4k-sft/export/execution','--captures','/tmp/onnax-qwen35-complete-context4k-block-captures','--block','decoder_3_4','--provider','mlx','--library',lib,'--warmup','10','--samples','100','--checkpoint','--recompute-attention']

 if mode=='zero':env['ONNXRUNTIME_EP_MLX_ATTENTION_VJP_DENSE_LIMIT_BYTES']='0'
 command=[python,'-m','examples.huggingface.guard','--output-dir',str(root/(mode+'-guard')),'--',*base,'--output',str(root/(mode+'.json'))]
 print(json.dumps({'phase':mode}),flush=True)
 with open(root/(mode+'.log'),'w') as log:child=subprocess.Popen(command,cwd='/tmp/onnax-kernel-contracts',env=env,stdout=log,stderr=subprocess.STDOUT)

print((root/'comparison.json').read_text(),flush=True)
