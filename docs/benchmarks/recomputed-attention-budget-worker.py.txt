import onnax

LIB='/tmp/onnax-kernel-contracts-runtime/libonnxruntime_mlx_ep.dylib'

p=argparse.ArgumentParser();p.add_argument('--mode',choices=['reference','dense','tiled','recompute'],required=True);p.add_argument('--directory',required=True);p.add_argument('--library',default=LIB);p.add_argument('--sequence-length',type=int,default=4096);p.add_argument('--head-dimension',type=int,default=128);p.add_argument('--query-heads',type=int,default=16);p.add_argument('--kv-heads',type=int,default=8);p.add_argument('--warmup',type=int,default=3);p.add_argument('--samples',type=int,default=10);p.add_argument('--label');a=p.parse_args();root=Path(a.directory)
S=a.sequence_length;D=a.head_dimension;HQ=a.query_heads;HK=a.kv_heads;LIB=a.library;SCALE=float(np.float32(1/np.sqrt(D)));label=a.label or a.mode
assert S>0 and D>0 and HQ>0 and HK>0 and HQ%HK==0 and a.samples>0 and a.warmup>=0
def forward():
 q=ir.val('Q',dtype=ir.DataType.FLOAT,shape=[1,S,HQ*D]);k=ir.val('K',dtype=ir.DataType.FLOAT,shape=[1,S,HK*D]);v=ir.val('V',dtype=ir.DataType.FLOAT,shape=[1,S,HK*D]);m=ir.val('Mask',dtype=ir.DataType.BOOL,shape=[1,1,1,S]);y=ir.val('Y',dtype=ir.DataType.FLOAT,shape=[1,S,HQ*D])
 return ir.Model(ir.Graph([q,k,v,m],[y],nodes=[ir.node('Attention',[q,k,v,m],{'q_num_heads':HQ,'kv_num_heads':HK,'is_causal':1,'scale':SCALE},outputs=[y])],opset_imports={'':24},name='attention'),ir_version=10)
register_library(LIB)
if a.mode=='reference':
 import torch
 torch.set_num_threads(1)
 rng=np.random.default_rng(42)
 feeds={name:rng.normal(0,scale,shape).astype(np.float32) for name,scale,shape in [('Q',1,(1,S,HQ*D)),('K',1,(1,S,HK*D)),('V',.1,(1,S,HK*D)),('G',.01,(1,S,HQ*D))]};feeds['Mask']=np.ones((1,1,1,S),bool)
 original=forward();onnx.checker.check_model(ir.to_proto(original),full_check=True)
 feeds['SavedY']=session(original,enable_compile=False).run(None,{k:feeds[k] for k in ('Q','K','V','Mask')})[0]
 leaves=[torch.tensor(feeds[name],device='mps',requires_grad=True) for name in ('Q','K','V')]
 q,k,v=[t.reshape(1,S,h,D).transpose(1,2) for t,h in zip(leaves,(HQ,HK,HK),strict=True)]
 y=torch.nn.functional.scaled_dot_product_attention(q,k.repeat_interleave(HQ//HK,dim=1),v.repeat_interleave(HQ//HK,dim=1),is_causal=True,scale=SCALE).transpose(1,2).reshape(1,S,HQ*D)
 g=torch.tensor(feeds['G'],device='mps');grads=torch.autograd.grad(y,leaves,g);torch.mps.synchronize()
 refs={name:value.detach().cpu().numpy() for name,value in zip(('dQ','dK','dV'),grads,strict=True)}
 np.testing.assert_allclose(feeds['SavedY'],y.detach().cpu().numpy(),rtol=1e-4,atol=1e-6)
 np.savez(root/'feeds.npz',**feeds);np.savez(root/'reference.npz',**refs)
 print(json.dumps({'phase':'independent_reference_complete'}),flush=True)
else:
 from examples.huggingface.long_context import memory_limit
 memory_limit(LIB,20*1024**3)
 data=dict(np.load(root/'feeds.npz'));refs=dict(np.load(root/'reference.npz'))
 joint=onnax.vjp(forward(),inputs=['Q','K','V'],outputs=['Y']);meta=json.loads(joint.graph.metadata_props['onnax.autodiff.interface'])
 joint=onnax.FuseAttentionVJPPass(recompute=a.mode=='recompute')(joint).model
 selected={value.name:value for value in joint.graph.outputs};joint.graph.outputs.clear();joint.graph.outputs.extend(selected[name] for name in meta['derivative_outputs'])
 if a.mode!='recompute':
  primal=next(node for node in joint.graph if node.op_type=='Attention' and node.domain=='').outputs[0]
  saved=ir.val('SavedY',dtype=primal.dtype,shape=primal.shape);joint.graph.inputs.append(saved);primal.replace_all_uses_with(saved)
 common.RemoveUnusedNodesPass()(joint)
 onnx.checker.check_model(ir.to_proto(joint),full_check=True)
 calls=[n.op_type for n in joint.graph if n.domain=='pkg.onnax'];assert len(calls)==1,calls
 feed={name:data[name] for name in ('Q','K','V','Mask')};feed[meta['seeds'][0]]=data['G']
 if a.mode!='recompute':feed['SavedY']=data['SavedY']
 runtime=session(joint,enable_compile=False);actual=runtime.run(None,feed)
 errors={}
 for (name,expected),value in zip(refs.items(),actual,strict=True):
  errors[name]={'maximum_absolute_error':float(np.max(np.abs(value-expected))),'violations':int(np.count_nonzero(~np.isclose(value,expected,rtol=1e-4,atol=1e-6)))}
 np.savez(root/(label+'-outputs.npz'),**dict(zip(refs,actual,strict=True)))
 samples=[]
 for i in range(a.warmup+a.samples):

  if i>=a.warmup:samples.append(elapsed)

 (root/(label+'.json')).write_text(json.dumps(report,indent=2)+'\n')
 print(json.dumps({'mode':a.mode,'timings_ms':report['timings_ms'],'errors':errors}),flush=True)
 assert not any(row['violations'] for row in errors.values()),errors
